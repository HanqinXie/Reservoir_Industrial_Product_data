w-richtext"><h2>HydroATLAS (version 1)</h2></div><div class="rich-text-block---regular-text w-richtext"><p>HydroATLAS offers a global compendium of hydro-environmental characteristics for all sub-basins of HydroBASINS, all river reaches of HydroRIVERS, and all lake polygons of HydroLAKES.</p><p>The HydroATLAS database is divided into three distinct sub-datasets: BasinATLAS, RiverATLAS, and LakeATLAS which represent sub-basin delineations (polygons), the river network (lines), and lake shorelines (polygons), respectively. In total, HydroATLAS contains 1.0 million sub-basins, 8.5 million river reaches, and 1.4 million lakes.</p><p>HydroATLAS has been created by compiling and re-formatting a wide range of hydro-environmental attributes derived from existing global datasets in a consistent and organized manner. The resulting data compendium offers attributes grouped in seven categories: hydrology; physiography; climate; land cover &amp; use; soils &amp; geology; and anthropogenic influences. For each of the three sub-datasets, HydroATLAS contains 56 hydro-environmental variables, partitioned into 281 individual attributes.</p><p>‍</p><p>The HydroATLAS database is distributed in large file sizes due to the enriched attribute information. Users who only need geometric information and digital vector maps of sub-basin boundaries, river network lines, and lake shorelines may prefer to download the <a href="/products/hydrobasins">HydroBASINS</a>, <a href="/products/hydrorivers">HydroRIVERS</a>, or <a href="/products/hydrolakes">HydroLAKES</a> products instead.     const boundNewsletterForms = new WeakSet();

    function escapeSel(id) {
      if (window.CSS && typeof window.CSS.escape === "function") return window.CSS.escape(id);
      return id.replace(/["\\#.;:[\],>+~*'()]/g, "\\$&");
    }

    function fieldInForm(form, elementId) {
      if (!form || !elementId) return null;
      return form.querySelector("#" + escapeSel(elementId));
    }

    function trimValue(el) {
      if (!el || typeof el.value !== "string") return "";
      return el.value.trim();
    }

    function collectExtraFields() {
      const out = {};
      for (const row of EXTRA_SUBSCRIBE_FIELDS) {
        const el = document.getElementById(row.elementId);
        if (!el) continue;
        const t = row.type || "string";
        if (t === "checkbox") {
          out[row.jsonKey] = Boolean(el.checked);
        } else {
          const v = trimValue(el);
          if (v !== "") out[row.jsonKey] = v;
        }
      }
      return out;
    }

    function buildSubscribeBody(email, name, signupSource) {
      return Object.assign(
        {
          email: email,
          name: name,
          signup_source: signupSource,
          tool_interest: false,
          message: "",
        },
        collectExtraFields()
      );
    }

    async function postSubscribe(body) {
      const headers = { "Content-Type": "application/json" };
      if (SUBSCRIBE_SECRET) headers["X-Subscribe-Secret"] = SUBSCRIBE_SECRET;
      return fetch(PROXY_SUBSCRIBE_URL, {
        method: "POST",
        headers: headers,
        body: JSON.stringify(body),
      });
    }

    function bindNewsletterForm(form, signupSource) {
      if (!form || boundNewsletterForms.has(form)) return;
      const emailEl = fieldInForm(form, NEWSLETTER_EMAIL_ID);
      const nameEl = fieldInForm(form, NEWSLETTER_NAME_ID);
      if (!emailEl) return;

      boundNewsletterForms.add(form);

      form.addEventListener("submit", async function (event) {
        event.preventDefault();

        const email = trimValue(emailEl);
        const name = nameEl ? trimValue(nameEl) : "";
        if (!email) {
          console.warn("Newsletter signup: missing email (#" + NEWSLETTER_EMAIL_ID + ").");
          return;
        }

        const body = buildSubscribeBody(email, name, signupSource);

        try {
          const response = await postSubscribe(body);
          if (!response.ok) {
            console.warn("Newsletter signup failed:", await response.text());
            return;
          }
          form.reset();
        } catch (err) {
          console.warn("Newsletter signup failed:", err);
        }
      });
    }

    for (const cfg of NEWSLETTER_FORMS) {
      const form = document.getElementById(cfg.formId);
      if (form) bindNewsletterForm(form, cfg.signupSource);
    }

    // Contact form: opt-in checkbox then native Webflow submit
    const contactForm = document.getElementById("wf-form-contact-general");
    const contactNameField = document.getElementById("First-Name-2");
    const contactEmailField = document.getElementById("Email-2");
    const optinCheckbox = document.getElementById("checkbox-2");
    let contactSubmitting = false;

    if (contactForm && contactEmailField && optinCheckbox) {
      contactForm.addEventListener("submit", async function (event) {
        if (contactSubmitting) return;
        if (!optinCheckbox.checked) return;

        event.preventDefault();

        const email = trimValue(contactEmailField);
        const name = contactNameField ? trimValue(contactNameField) : "";

        if (email) {
          try {
            const response = await postSubscribe(buildSubscribeBody(email, name, "contact"));
            if (!response.ok) {
              console.warn("Newsletter opt-in failed:", await response.text());
            }
          } catch (err) {
            console.warn("Newsletter opt-in failed:", err);
          }
        }

        contactSubmitting = true;
        contactForm.requestSubmit();
      });
    }
  });
